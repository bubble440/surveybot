function RepeatHeaders(r, qid, divId){
  var mediaQuery = window.matchMedia("only screen and (min-width: 641px)");
  var thisQuestion = document.getElementById(qid);
  var header = thisQuestion.getElementsByClassName(divId+"-answer--fake-for-panel")[0];
  var parent = header.parentNode; 
  var allRows = thisQuestion.getElementsByClassName("cf-grid-layout__row");
  var labels = document.getElementsByClassName('cf-gb-grid-answer__scale-label-panel')
  if(mediaQuery.matches){
    for(var i = 1; i < allRows.length; i++) {
      var headerCopy = header.cloneNode(true);
      var repeatLabels = headerCopy.getElementsByClassName(divId+"-answer__label")
      for(var j = 0; j < repeatLabels.length; j++){
        repeatLabels[j].id = "";
      }
      
      if(i % (r+1) == 0) {
        var inserted = parent.insertBefore(headerCopy, allRows[i]);
        //If this is a thermometer then also need to make sure the scale end texts show
        //Just need this for Edge - in other browsers the scale ends get repeated when they're cloned
        if(divId == 'cf-gb-grid'){
          labels[i].style = "display: flex !important";
        }
      }
    }
  }
}

//Move other specify placeholder text into the answer text
function moveOtherSpecify() {
  var mobile = window.matchMedia("(max-width: 640px)");
  var addGridHeaderStyle = false;
  var addMultiHeaderStyle = false;
  var hideOtherScale = false;
  var add3dGridAnswerStyle = false;
  //get placeholder text
  var otherSpecify = document.getElementsByClassName("cf-answer-button__other-input"); //multis and singles
  if(otherSpecify.length == 0) {
    //For an other specify on a grid or multi-grid
    otherSpecify = document.getElementsByClassName("cf-grid-answer__other");
    //If this is a grid (not a multi grid) and it's mobile, then we also need to style the heading
    var multiGrid = document.getElementsByClassName('cf-grid-answer--multi');
    if(mobile.matches) {  
      addMultiHeaderStyle = true;
      if(multiGrid.length == 0) {
        addGridHeaderStyle = true;
      }
    } else {
      //If this is a multi grid, then don't want to do this for the scale items on desktop
      if(multiGrid.length > 0) {
        hideOtherScale = true;
      }
    }
  }
  if(otherSpecify.length == 0 && !mobile.matches) {
    //For an other specify on a 3D grid
    otherSpecify = document.getElementsByClassName("cf-grid-3d-desktop__answer-other-input");
  }
  if(otherSpecify.length == 0 && mobile.matches) {
    //For an other specify on a grid or 3D grid on mobile
    otherSpecify = document.getElementsByClassName("cf-multi-answer__other-input");
    add3dGridAnswerStyle = true;
    //Also need to do this if we have a grid inside a 3D grid
    var gridRow = document.getElementsByClassName('cf-grid-layout__row-text');
    for(var k = 0; k < gridRow.length; k++) {
      if(gridRow[k].querySelector('.cf-text-box') != null) {
        var others = gridRow[k].querySelector('.cf-text-box');
        var placeholder = others.getAttribute("placeholder");
        var answerButton = gridRow[k];
        // create answer button text to match with other answers
        var answerButtonText = document.createElement("div");
        // and give it some content
        var answerButtonTextContent = document.createTextNode(placeholder); 
        // add the text node to the newly created div
        answerButtonText.appendChild(answerButtonTextContent);  
        // add the newly created element and its content into the DOM 
        answerButton.insertBefore(answerButtonText, others)
        answerButtonText.classList.add('cf-answer-button__text');
        //Make the input go below the text by changing flex-direction on this node to column
        answerButton.classList.add('flex-column');
        answerButton.classList.add('grey-background');
        answerButton.style.alignSelf = 'stretch';
        others.style.width = '200px';
      }
    }
  }
  for(var i = 0; i < otherSpecify.length; i++){
    var placeholder = otherSpecify[i].getAttribute("placeholder");
    //We can't use the class to find the answer button because there are loads and we don't know which it is
    //But it will be the parent of the other specify
    var answerButton = otherSpecify[i].parentElement;
    // create answer button text to match with other answers
    var answerButtonText = document.createElement("div"); 
    // and give it some content 
    var answerButtonTextContent = document.createTextNode(placeholder); 
    // add the text node to the newly created div
    answerButtonText.appendChild(answerButtonTextContent);  
    // add the newly created element and its content into the DOM 
    answerButton.insertBefore(answerButtonText, otherSpecify[i])
    answerButtonText.classList.add('cf-answer-button__text');
    //Make the input go below the text by changing flex-direction on this node to column
    answerButton.classList.add('flex-column');
    if(addGridHeaderStyle) {
      answerButton.classList.add('grey-background');
    }
    if(addGridHeaderStyle || addMultiHeaderStyle) {
      answerButton.style.alignSelf = 'stretch';
      otherSpecify[i].style.width = '200px';
    }
    if(mobile.matches && !addMultiHeaderStyle) {
      answerButton.style.fontWeight = '400';
      otherSpecify[i].style.paddingLeft = '40px';
    }
    if(mobile.matches && add3dGridAnswerStyle) {
      answerButton.style.fontWeight = '400';
      answerButton.style.paddingLeft = '40px';
    }
    if(hideOtherScale) {
      var scaleButton = document.getElementsByClassName('cf-grid-answer__scale-item');
      for(var j = 0; j < scaleButton.length; j++) {
        var childText = scaleButton[j].querySelector('.cf-answer-button__text');
        if(childText != null) {
          childText.style.display = 'none';
        }
      }
    }
    //On 3d grids on desktop need to right align the other specify
    var grid3d = document.getElementsByClassName('cf-grid-3d-desktop__content');
    if(grid3d.length > 0 && !mobile.matches){
      answerButton.style.textAlign = 'right';
    }
  }
}

//Change image to grey on disabled answers on grids
function disabledImage() {
 var disabled = document.getElementsByClassName('answerDisabled');
 for(var i = 0; i < disabled.length; i++) {
   //Get background image link
   var style = disabled[i].currentStyle || window.getComputedStyle(disabled[i], false);
   var image = style.backgroundImage.slice(4, -1).replace(/"/g, "");
   //Get just the last bit (eg radio-purple.png or checkbox-purple.png)
   var bits = image.split("/");
   var lastBit = bits[bits.length-1];
   //Now get the bit before the hyphen - so will get just radio or checkbox
   var radioCheck = lastBit.split("-");
   if(radioCheck[0] == "radio") {
     disabled[i].style.backgroundImage = "url(https://www.qubiq-surveys.com/LAYOUTS/2019/imgs/radio-grey.png)";
   }
   if(radioCheck[0] == "checkbox") {
     disabled[i].style.backgroundImage = "url(https://www.qubiq-surveys.com/LAYOUTS/2019/imgs/checkbox-grey.png)";
   }
 }
}

//Disable answers functions
function chooseDisabledAnswers(type, answers) {
  if(type == 'Grid') {
    return answers;
  }
  if(type == 'Grid3d') {
    var newAnswers = [];
    for(var a=0; a<answers.length; a++) {
      newAnswers.push('desktop_' + answers[a]);
      newAnswers.push('mobile_' + answers[a]);
    }
    return newAnswers;
  }
}

function disableAnswer(elementID){
  if( document.getElementById(elementID) != null ) {
    document.getElementById(elementID).classList.add("answerDisabled");
    document.getElementById(elementID).style.backgroundImage = "url(https://www.qubiq-surveys.com/LAYOUTS/2019/imgs/radio-grey.png)"; 
  }
}

//On load functions - Nexus survey tester and multi-column flex fix 
window.addEventListener('load', function(e) {
  
  // Nexus Survey Tester - Code that runs only with test links
  var isProd = document.getElementById("cstm_isProd").value;
  var cellData = document.getElementById("cstm_cell").value || "n/a";
  var rotationData = document.getElementById("cstm_rotation").value || "n/a";
  var langData = document.getElementById("cstm_language").value || "n/a";
  var marketData = document.getElementById("cstm_market").value.replace(/<em>NO RESPONSE<\/em>/gi, "n/a");
  
  if(isProd=="False"){
    //var questions = document.getElementsByClassName('cf-question'); // get all questions on page
    //Get question data from the ConfirmIt API 
    var qnId = window.Confirmit.page.questions[0];
   
    var qnIdData = {
      type: "SmartTesterPostMessage",
      questionId: qnId.id,
      rotation: rotationData,
      cell: cellData,
      language: langData,
      market: marketData
    }
   
   if( qnId ){ // if there are questions     
     window.top.postMessage(qnIdData, "*"); // send to SmartTester
   }
  } 
  
  
  //Update flex on multi-columns if they contain images
  //Can only do this once it's loaded because otherwise it hasn't added multi-columns yet
  var multiColumns = document.querySelector('.multi-columns');
  var listItem = document.querySelectorAll('.cf-list__item');
  if(multiColumns) {
    var images = document.querySelectorAll('img');
    if(images.length > 0){
      var imageWidths = [];
      //Find any img tags within multiColumns
      //If there, then set flex-grow to 0 on multi-columns
      for(var i=0; i<images.length; i++){
        if(multiColumns.contains(images[i])){
          for(var j=0; j<listItem.length; j++){
            listItem[j].style.flexGrow = 0;
          }
          //If it's a highslide then either .smallImg will have a width set, in which case we want that width
          //Or it will be auto in which case we'll get the display width
          //If it's not a highslide, then we want the original image width (as it might be shrunk)
          //If it's a highslide, the image will have a class of smallImg
          if(images[i].className == 'smallImg'){
            var smallImg = document.querySelector('.smallImg');
            var style = getComputedStyle(smallImg);

            if(style.width != '') {
              imageWidths.push(parseFloat(style.width.match(/\d*\.*\d*/g).join('')));
            } else {
              imageWidths.push(images[i].clientWidth);
            }
          } else {
            imageWidths.push(images[i].naturalWidth);
          }
          var allSameWidth = true;
          var widestImage = imageWidths[0];
          for(var w=1; w<imageWidths.length; w++){
            if(imageWidths[w] != imageWidths[w-1]){
              allSameWidth = false;
              if(imageWidths[w] > widestImage){
                widestImage = imageWidths[w];
              }
            }
          }
          //If they're the same then set flex-basis to auto
          //Otherwise set flex-basis to widest image + 50px
          var flexWidth = widestImage + 50;
          for(n=0; n<listItem.length; n++){
            if(allSameWidth){
              listItem[n].style.flexBasis = 'auto';
            } else {
              listItem[n].style.flexBasis = flexWidth + 'px';
            }
          }
        } //if multiColumns contains images
      } //images for loop
    } //images length
  } //if multiColumn
});
